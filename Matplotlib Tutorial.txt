#Scatter plota

import matplotlib.pyplot as plt
import numpy as np

x = np.random.randint(100, size=50)
y = np.random.randint(100, size=50)
colors = np.random.randint(100, size=50)
sizes = np.random.randint(500, size=50)

plt.scatter(x, y, c=colors, s=sizes, alpha=0.5, cmap='viridis')

plt.colorbar()
plt.title("Scatter Plot Example")

plt.show()

# Histogram

# Pie Chart

import matplotlib.pyplot as plt

y = [35, 25, 25, 15]
labels = ["Apples", "Bananas", "Cherries", "Dates"]
explode = [0.2, 0, 0, 0]
colors = ["red", "yellow", "pink", "green"]

plt.pie(y, labels=labels, explode=explode, colors=colors, shadow=True)

plt.title("Fruit Distribution")
plt.legend(title="Fruits")

plt.show()






#Count Plot (Univariate Categorical)

import matplotlib.pyplot as plt
from collections import Counter

data = ["A", "B", "A", "C", "B", "A"]

count = Counter(data)

plt.title("Count Plot")
plt.xlabel("Category")
plt.ylabel("Count")

plt.show()

#stacked bar chart

import matplotlib.pyplot as plt

department = ["HR", "IT", "Sales"]

male = [20, 30, 40]
female = [15, 25, 35]

plt.bar(department, male, label="Male")

plt.bar(department,
        female,
        bottom=male,
        label="Female")

plt.title("Employee Distribution")

plt.xlabel("Department")
plt.ylabel("Employees")

plt.legend()

plt.show()

# Grouped Bar Plot

import matplotlib.pyplot as plt
import numpy as np

department = ["HR", "IT", "Sales"]

male = [20, 30, 40]
female = [15, 25, 35]

x = np.arange(len(department))

width = 0.3

plt.bar(x - width/2,
        male,
        width,
        label="Male")

plt.bar(x + width/2,
        female,
        width,
        label="Female")

plt.xticks(x, department)

plt.xlabel("Department")
plt.ylabel("Employees")

plt.title("Grouped Bar Plot")

plt.legend()

plt.show()





# Multiple Line Plot

         label="Product A")

         sales_B,
         label="Product B")

plt.xlabel("Month")
plt.ylabel("Sales")

plt.title("Sales Comparison")

plt.legend()

plt.show()
