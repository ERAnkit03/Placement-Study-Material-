# Pandas - Working with Text Data

df = pd.DataFrame({
    "name": [" Ankit ", "RAHUL", "sneha"],
    "email": ["ankit@gmail.com", "rahul@yahoo.com", "sneha@outlook.com"]
})


#1. String Transformation (Change Text)

df["name"].str.lower()     # convert to lowercase
df["name"].str.upper()     # uppercase
df["name"].str.replace("a","A")  # replace


#2. String Trimming (Remove Extra Spaces)

df["name"].str.strip()   # remove spaces both sides
df["name"].str.lstrip()  # left side
df["name"].str.rstrip()  # right side

#3. Concatenation (Join Text)

df["full"] = df["first"].str.cat(df["last"], sep=" ")

#4. Padding (Adjust Length)

df["num"].str.zfill(5)

#5. Searching (Find Text)

df["name"].str.contains("an")   # True/False
df["name"].str.count("a")       # count letters
df["name"].str.startswith("A")

#6. Splitting (Break Text)

df["email"].str.split("@")

#7. Filtering (Check Conditions)

df["name"].str.isalpha()   # only letters?
df["num"].str.isdigit()    # only numbers?
df["text"].str.len()       # length

#8. Miscellaneous (Other Operations)

df["name"].str.repeat(2)      # repeat text
df["name"].str.slice(0,3)     # substring
df["name"].str.endswith("t")  # check ending

# Pandas - Comparing Categorical Data

print(s == "Low")

s = pd.Series(["Low", "High", "Medium"])

s = s.cat.set_categories(["Low", "Medium", "High"], ordered=True)

s2 = pd.Series(["Low", "Low"], dtype="category")

# Pandas - Dropping Missing Values

import pandas as pd
import numpy as np

data = {
    "Name": ["Ajay", "Krishna", "Deepak", "Swati", "Riya"],
    "Age": [20, 21, np.nan, 19, np.nan],
    "Marks": [85, np.nan, 90, np.nan, np.nan],
    "City": ["Pune", "Mumbai", "Delhi", None, "Pune"]
}

df = pd.DataFrame(data)
print("Original DataFrame:\n", df)


#Drop Rows with ANY Missing Values
df1 = df.dropna()
print(df1)

#Drop Rows Only if ALL Values Missing
df2 = df.dropna(how='all')
print(df2)

#Keep Rows with Minimum Data (thresh)
df3 = df.dropna(thresh=3)
print(df3)

#Drop Based on Important Columns
df4 = df.dropna(subset=["Age", "Marks"])
print(df4)

#Drop Columns with Missing Values
df5 = df.dropna(axis=1)
print(df5)
